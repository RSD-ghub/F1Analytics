# up whatever is newest that day, which is how bcrypt 5.0 and FastAPI 0.16x
# arrived unannounced. Docker builds and local installs both read it:
#
#   pip install -c constraints.txt -r services/core-api/requirements.txt
#   pip install -c constraints.txt -r requirements-dev.txt
#
# To upgrade on purpose: change a range in a requirements.txt (or leave them),
# then regenerate and run the tests:
#
#   scripts/lock_requirements.sh
